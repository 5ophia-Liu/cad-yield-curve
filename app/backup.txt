
# setup curve endpoint
@app.get("/curve")
def get_curve(start_date: str, end_date: str):
    try: 
        yield_df = extract_yield_dataframe(fetch_benchmark_yields(start_date, end_date))

        history = []
        for _, row in yield_df.iterrows():
            date_val = row["date"]
            yields = row.drop(labels=["date"]).to_dict()
            curve = build_curve(yields, ql_date(date_val), calendar, daycount_convention) 
            print("TYPE OF CURVE:", type(curve))

            curve_dict = curve_to_dict(curve, calendar, ql_date(date_val)) # Convert to dict
            history.append({
                "date": date_val,
                "curve": curve_dict
            })
        return history
    except Exception as e:
        import traceback
        traceback.print_exc()
        raise HTTPException(status_code=400, detail=str(e))
